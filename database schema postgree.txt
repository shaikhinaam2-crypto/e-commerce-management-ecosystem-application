-- Enable UUID extension for secure, unique identifiers
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- Table 1: Landing Page Hero & Core Content
CREATE TABLE IF NOT EXISTS public_landing_content (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    hero_title VARCHAR(255) NOT NULL,
    hero_subtitle TEXT NOT NULL,
    cta_button_text VARCHAR(100) DEFAULT 'Start 14-Day Free Trial',
    cta_button_link VARCHAR(255) DEFAULT '/signup',
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Table 2: Video Section Content (Supports YouTube Embeds)
CREATE TABLE IF NOT EXISTS landing_videos (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    title VARCHAR(255) NOT NULL,
    description TEXT,
    youtube_url VARCHAR(500) NOT NULL,
    youtube_video_id VARCHAR(100) NOT NULL, -- Extracted video ID (e.g., 'dQw4w9WgXcQ')
    display_order INT DEFAULT 1,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Table 3: Ecosystem Modules Overview (Showcased on Landing Page)
CREATE TABLE IF NOT EXISTS landing_module_showcase (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    module_title VARCHAR(150) NOT NULL,
    short_description TEXT NOT NULL,
    icon_name VARCHAR(100), -- FontAwesome / Lucide icon identifier
    badge_tag VARCHAR(50) DEFAULT 'Included in Trial',
    display_order INT DEFAULT 1,
    is_active BOOLEAN DEFAULT TRUE
);

-- Table 4: Dynamic Pricing & Trial Offer Config
CREATE TABLE IF NOT EXISTS landing_pricing_plans (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    plan_name VARCHAR(100) NOT NULL,
    price_monthly NUMERIC(10, 2) DEFAULT 0.00,
    currency VARCHAR(10) DEFAULT 'USD',
    trial_days INT DEFAULT 14,
    features_list JSONB NOT NULL, -- Array of feature strings
    is_popular BOOLEAN DEFAULT FALSE,
    display_order INT DEFAULT 1,
    is_active BOOLEAN DEFAULT TRUE
);
